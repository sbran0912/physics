	screenWidth := int32(1200)
	screenHeight := int32(800)
	rl.SetConfigFlags(rl.FlagVsyncHint) // rl.FlagFullscreenMode |
	rl.InitWindow(screenWidth, screenHeight, "Physics - raylib-go")
	defer rl.CloseWindow()

	a := CreateBox(200, 100, 150, 150)
	b := CreateBox(250, 270, 120, 150)
	//a.rotate(-0.5)

	//isColliding, mtv, contacts := SATCollision(a.vertices[:], b.vertices[:])
	//fmt.Println(isColliding, mtv, contacts)

	for !rl.WindowShouldClose() {

		rl.BeginDrawing()
		rl.ClearBackground(rl.DarkBlue)

		_, mtv, contacts := SATCollision(a, b)
		//fmt.Println(contacts)

		a.FollowMouse()

		b.Draw(rl.White, 2)

		for _, c := range contacts {
			rl.DrawCircleV(c, 5, rl.Black)
		}
		rl.EndDrawing()
	}
}

type ShapeType int

const (
	CircleShape ShapeType = iota
	PolygonShape
)

type Shape interface {
	draw()
	rotate()
	FollowMouse()
}

type BasicShape struct {
	typ         ShapeType
	location    rl.Vector2
	velocity    rl.Vector2
	angVelocity float32
	accel       rl.Vector2
	angAccel    float32
	mass        float32
	inertia     float32
}

type Polygon struct {
	basic    BasicShape
	vertices []rl.Vector2
}

type Circle struct {
	basic       BasicShape
	radius      int
	orientation rl.Vector2
}

func CreateBox(x float32, y float32, w float32, h float32) Polygon {
	box := Polygon{
		basic:    BasicShape{typ: PolygonShape, location: rl.Vector2{x + w/2, y + h/2}},
		vertices: []rl.Vector2{{x, y}, {x + w, y}, {x + w, y + h}, {x, y + h}},
	}
	return box
}

func (poly Polygon) Draw(c rl.Color, thick float32) {
	n := len(poly.vertices)
	for i := range poly.vertices {
		rl.DrawLineEx(poly.vertices[i], poly.vertices[(i+1)%n], thick, c)
	}
	rl.DrawCircleV(poly.basic.location, 5, c)
}

func (poly *Polygon) rotate(angle float32) {

	for i := range poly.vertices {
		relativePos := rl.Vector2Subtract(poly.vertices[i], poly.basic.location)
		rotatedPos := rl.Vector2Rotate(relativePos, angle)
		poly.vertices[i] = rl.Vector2Add(rotatedPos, poly.basic.location)
	}
}

func (poly *Polygon) FollowMouse() {
	mouse := rl.GetMousePosition()

	// Differenz zwischen alter Position und Maus
	delta := rl.Vector2Subtract(mouse, poly.basic.location)

	// Alle Vertices verschieben
	for i := range poly.vertices {
		poly.vertices[i] = rl.Vector2Add(poly.vertices[i], delta)
	}

	// Location aktualisieren
	poly.basic.location = mouse
}

// Projektionsbereich auf einer Achse
func projectPolygon(vertices []rl.Vector2, axis rl.Vector2) (min, max float32) {
	min = rl.Vector2DotProduct(vertices[0], axis)
	max = min
	for _, vertex := range vertices[1:] {
		proj := rl.Vector2DotProduct(vertex, axis)
		if proj < min {
			min = proj
		}
		if proj > max {
			max = proj
		}
	}
	return min, max
}

// Alle Normalenachsen eines Polygons
func getAxes(vertices []rl.Vector2) []rl.Vector2 {
	axes := []rl.Vector2{}
	n := len(vertices)
	for i := range n {
		edge := rl.Vector2Normalize(rl.Vector2Subtract(vertices[(i+1)%n], vertices[i]))
		axes = append(axes, rl.Vector2{-edge.Y, edge.X})
	}
	return axes
}

func pointInPolygon(point rl.Vector2, vertices []rl.Vector2) bool {
	n := len(vertices)
	for i := range n {
		edge := rl.Vector2Subtract(vertices[(i+1)%n], vertices[i])
		edge_perp := rl.Vector2{-edge.Y, edge.X}
		toPoint := rl.Vector2Subtract(point, vertices[i])

		if rl.Vector2DotProduct(edge_perp, toPoint) < 0 {
			return false
		}
	}
	return true
}

func findContactPoints(polyA_vertices, polyB_vertices []rl.Vector2) []rl.Vector2 {
	contacts := []rl.Vector2{}

	// 1. Checke, welche Ecken von A in B liegen
	for _, p := range polyA_vertices {
		if pointInPolygon(p, polyB_vertices) {
			contacts = append(contacts, p)
		}
	}

	// 2. Checke, welche Ecken von B in A liegen

	for _, p := range polyB_vertices {
		if pointInPolygon(p, polyA_vertices) {
			contacts = append(contacts, p)
		}
	}

	return contacts
}

func findReferenceEdge(vertices []rl.Vector2, normal rl.Vector2) (p1, p2 rl.Vector2) {
	bestDot := float32(-math.MaxFloat32)
	n := len(vertices)
	for i := range n {
		a := vertices[i]
		b := vertices[(i+1)%n]

		edge := rl.Vector2Normalize(rl.Vector2Subtract(b, a))
		edgeNormal := rl.Vector2{-edge.Y, edge.X}

		d := rl.Vector2DotProduct(edgeNormal, normal)
		if d > bestDot {
			bestDot = d
			p1 = a
			p2 = b
		}
	}
	return p1, p2
}

func projectPointOntoEdge(p, a, b rl.Vector2) rl.Vector2 {
	ab := rl.Vector2Subtract(b, a)
	t := rl.Vector2DotProduct(rl.Vector2Subtract(p, a), ab) /
		rl.Vector2DotProduct(ab, ab)

	// Clamp auf Segment
	if t < 0 {
		t = 0
	} else if t > 1 {
		t = 1
	}

	return rl.Vector2Add(a, rl.Vector2Scale(ab, t))
}

func transferContactsToA(contacts []rl.Vector2, polyA_vertices []rl.Vector2, normal rl.Vector2) []rl.Vector2 {

	refA, refB := findReferenceEdge(polyA_vertices, normal)

	projected := []rl.Vector2{}
	for _, c := range contacts {
		p := projectPointOntoEdge(c, refA, refB)
		projected = append(projected, p)
	}
	return projected
}

// SAT-Kollisionstest
func SATCollision(polyA, polyB Polygon) (bool, rl.Vector2, []rl.Vector2) {
	smallestOverlap := float32(math.MaxFloat32)
	var smallestAxis rl.Vector2
	axes := append(getAxes(polyA.vertices), getAxes(polyB.vertices)...)

	for _, axis := range axes {
		minA, maxA := projectPolygon(polyA.vertices, axis)
		minB, maxB := projectPolygon(polyB.vertices, axis)

		overlap := float32(math.Min(float64(maxA), float64(maxB))) - float32(math.Max(float64(minA), float64(minB)))
		if overlap <= 0 {
			return false, rl.Vector2{}, nil
		}

		if overlap < smallestOverlap {
			smallestOverlap = overlap
			smallestAxis = axis
			// Richtung korrigieren
			dir := rl.Vector2Subtract(polyB.basic.location, polyA.basic.location)
			if rl.Vector2DotProduct(dir, axis) < 0 {
				smallestAxis = rl.Vector2Scale(axis, -1)
			}
		}
	}

	mtv := rl.Vector2Scale(smallestAxis, smallestOverlap)

	rawContacts := findContactPoints(polyA.vertices, polyB.vertices)

	contacts := transferContactsToA(rawContacts, polyA.vertices, rl.Vector2Scale(rl.Vector2Normalize(smallestAxis), -1))

	return true, mtv, contacts
}
